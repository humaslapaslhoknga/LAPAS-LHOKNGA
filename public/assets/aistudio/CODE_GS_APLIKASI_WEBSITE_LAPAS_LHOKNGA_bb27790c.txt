function doGet(e) {
  pastikanSetup_();
  const p = (e && e.parameter) || {};
  let route = '';
  if (p.p && /^[a-z0-9\-_]{1,60}$/i.test(p.p)) {
    route = '/' + p.p;
    if (p.s && /^[a-z0-9\-_]{1,120}$/i.test(p.s)) route += '/' + p.s;
  }
  const cfg = cfg_();
  const t = HtmlService.createTemplateFromFile('Index');
  t.bootJson = JSON.stringify({ baseUrl: ScriptApp.getService().getUrl(), route: route })
    .replace(/</g, '\\u003c').replace(/>/g, '\\u003e').replace(/\u2028/g, '\\u2028').replace(/\u2029/g, '\\u2029');
  const out = t.evaluate()
    .setTitle((cfg.Nama_Instansi || 'Lapas Kelas III Lhoknga') + ' - Website Resmi')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  try { out.setFaviconUrl(cfg.Logo_URL || DEFAULT_LOGO); } catch (err) {}
  return out;
}

function fail_(msg, code) { const e = new Error(msg); e.userMsg = msg; e.code = code || ''; throw e; }

function safe_(fn, detail) {
  try { return { ok: true, data: fn() }; }
  catch (e) {
    if (!e.userMsg) console.error(e && e.stack ? e.stack : e);
    return { ok: false, code: e.code || '', message: e.userMsg || (detail ? 'Kesalahan server: ' + String(e.message).slice(0, 220) : 'Terjadi kesalahan pada server. Silakan coba lagi.') };
  }
}

function withLock_(fn) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(20000); } catch (e) { fail_('Server sedang sibuk. Silakan coba lagi beberapa detik lagi.'); }
  try { return fn(); } finally { lock.releaseLock(); }
}

function now_() { return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm:ss'); }
function today_() { return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd'); }
function isDate_(s) { return /^\d{4}-\d{2}-\d{2}$/.test(String(s)) && !isNaN(Date.parse(s)); }
function addDays_(d, n) { const p = d.split('-').map(Number); return Utilities.formatDate(new Date(Date.UTC(p[0], p[1] - 1, p[2] + n)), 'UTC', 'yyyy-MM-dd'); }
function dow_(d) { const p = d.split('-').map(Number); return new Date(Date.UTC(p[0], p[1] - 1, p[2])).getUTCDay(); }
function int_(v, def) { const n = parseInt(v, 10); return isNaN(n) ? def : n; }
function str_(v, max) { return String(v === undefined || v === null ? '' : v).replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '').trim().slice(0, max || 500); }
function jsonSafe_(s, def) { try { const v = JSON.parse(s); return v === null || v === undefined ? def : v; } catch (e) { return def; } }
function esc_(s) { return String(s === undefined || s === null ? '' : s).replace(/[&<>"']/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]; }); }
function slugify_(s) { return String(s || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 80); }
function newId_(prefix) { return prefix + Utilities.getUuid().replace(/-/g, '').slice(0, 8).toUpperCase(); }
function randCode_(n) { const a = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; let s = ''; for (let i = 0; i < n; i++) s += a.charAt(Math.floor(Math.random() * a.length)); return s; }
function validUrl_(u, allowEmpty) { u = str_(u, 1000); if (!u) return allowEmpty === false ? fail_('URL wajib diisi.') : ''; if (!/^https:\/\/[^\s<>"']+$/i.test(u)) fail_('URL harus diawali https:// dan tidak boleh mengandung spasi.'); return u; }
function validEmail_(e) { return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(e); }
function normHp_(h) { h = String(h || '').replace(/[\s\-().]/g, ''); if (h.indexOf('+62') === 0) h = '0' + h.slice(3); else if (h.indexOf('62') === 0) h = '0' + h.slice(2); return h; }

let _ss = null;
function ss_() { if (!_ss) _ss = SpreadsheetApp.openById(SPREADSHEET_ID); return _ss; }
function sh_(name) {
  const book = ss_();
  let s = book.getSheetByName(name);
  if (!s) { s = book.insertSheet(name); initSheet_(s, name); }
  return s;
}
function initSheet_(s, name) {
  const H = SCHEMA[name];
  if (s.getMaxColumns() < H.length) s.insertColumnsAfter(s.getMaxColumns(), H.length - s.getMaxColumns());
  s.getRange(1, 1, 1, H.length).setNumberFormat('@').setValues([H]).setFontWeight('bold').setBackground(HEADER_BG[name] || '#01162a').setFontColor('#ffffff');
  if (s.getMaxRows() > 1) s.getRange(2, 1, s.getMaxRows() - 1, H.length).setNumberFormat('@');
  s.setFrozenRows(1);
}
function cell_(v) { if (v instanceof Date) return Utilities.formatDate(v, TZ, 'yyyy-MM-dd HH:mm:ss'); return v === null || v === undefined ? '' : String(v); }
function table_(name) {
  const H = SCHEMA[name], s = sh_(name), last = s.getLastRow();
  if (last < 2) return [];
  const v = s.getRange(2, 1, last - 1, H.length).getValues();
  const out = [];
  for (let i = 0; i < v.length; i++) {
    const o = { _row: i + 2 };
    for (let j = 0; j < H.length; j++) o[H[j]] = cell_(v[i][j]);
    if (o[H[0]] !== '') out.push(o);
  }
  return out;
}
function rowArr_(name, obj) { return SCHEMA[name].map(function (k) { return obj[k] === undefined || obj[k] === null ? '' : String(obj[k]); }); }
function addRow_(name, obj) {
  const s = sh_(name), n = SCHEMA[name].length, r = s.getLastRow() + 1;
  if (r > s.getMaxRows()) s.insertRowsAfter(s.getMaxRows(), 200);
  s.getRange(r, 1, 1, n).setNumberFormat('@').setValues([rowArr_(name, obj)]);
  return r;
}
function setRow_(name, row, obj) { sh_(name).getRange(row, 1, 1, SCHEMA[name].length).setNumberFormat('@').setValues([rowArr_(name, obj)]); }
function delRow_(name, row) { sh_(name).deleteRow(row); }
function clean_(o) { const c = {}; Object.keys(o).forEach(function (k) { if (k !== '_row') c[k] = o[k]; }); return c; }

function pastikanSetup_() {
  const props = PropertiesService.getScriptProperties();
  if (props.getProperty('SETUP_VERSION') === SETUP_VERSION) return;
  withLock_(function () {
    if (props.getProperty('SETUP_VERSION') === SETUP_VERSION) return;
    Object.keys(SCHEMA).forEach(function (name) {
      const book = ss_();
      let s = book.getSheetByName(name);
      if (!s) { s = book.insertSheet(name); }
      initSheet_(s, name);
    });
    seedPengaturan_(); seedUsers_(); seedHalaman_(); seedPelayanan_(); seedProgram_(); isiKontenBawaan_();
    bersihCache_();
    props.setProperty('SETUP_VERSION', SETUP_VERSION);
  });
}

function hash_(pw, salt) {
  let d = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + ':' + pw);
  const sb = Utilities.newBlob(salt).getBytes();
  for (let i = 0; i < HASH_ROUNDS; i++) d = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, d.concat(sb));
  return d.map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
}
function cekPasswordBaru_(p) {
  p = String(p || '');
  if (p.length < 8) fail_('Password baru minimal 8 karakter.');
  if (!/[A-Za-z]/.test(p) || !/\d/.test(p)) fail_('Password baru harus mengandung huruf dan angka.');
  if (p.length > 100) fail_('Password terlalu panjang.');
}

function auth_login(username, password) {
  return safe_(function () {
    username = str_(username, 60).toLowerCase(); password = String(password || '');
    if (!username || !password) fail_('Username dan password wajib diisi.');
    const cache = CacheService.getScriptCache(), fk = 'fail_' + username;
    const fails = int_(cache.get(fk), 0);
    if (fails >= 5) fail_('Terlalu banyak percobaan gagal. Silakan coba lagi dalam 15 menit.');
    const u = table_('Users').filter(function (r) { return r.Username.toLowerCase() === username; })[0];
    if (!u || u.Status !== 'Aktif' || hash_(password, u.Salt) !== u.Password_Hash) {
      cache.put(fk, String(fails + 1), 900);
      fail_('Username atau password salah, atau akun tidak aktif.');
    }
    cache.remove(fk);
    const token = Utilities.getUuid() + Utilities.getUuid().replace(/-/g, '');
    const sess = { username: u.Username, nama: u.Nama, role: u.Role, jabatan: u.Jabatan, iat: Date.now(), wajib: u.Wajib_Ganti === 'Ya' };
    cache.put('sess_' + token, JSON.stringify(sess), SESI_TTL);
    withLock_(function () { const r = table_('Users').filter(function (x) { return x.Username === u.Username; })[0]; if (r) { r.Terakhir_Login = now_(); setRow_('Users', r._row, r); } });
    log_(sess, 'Login', '');
    return { token: token, user: userPublik_(sess) };
  });
}

function auth_(token, roles, allowWajib) {
  token = String(token || '');
  const cache = CacheService.getScriptCache();
  const raw = token.length > 20 ? cache.get('sess_' + token) : null;
  if (!raw) fail_('Sesi Anda telah berakhir. Silakan masuk kembali.', 'SESI');
  const s = JSON.parse(raw);
  const rev = cache.get('rev_' + s.username);
  if (rev && Number(rev) > s.iat) { cache.remove('sess_' + token); fail_('Sesi Anda tidak berlaku lagi. Silakan masuk kembali.', 'SESI'); }
  if (s.wajib && !allowWajib) fail_('Anda wajib mengganti password terlebih dahulu.', 'GANTI');
  if (roles && roles.length && s.role !== R_SUPER && roles.indexOf(s.role) < 0) fail_('Anda tidak memiliki hak akses untuk fitur ini.', 'AKSES');
  cache.put('sess_' + token, raw, SESI_TTL);
  return s;
}
function cabutSesi_(username) { CacheService.getScriptCache().put('rev_' + username, String(Date.now()), SESI_TTL); }

function auth_me(token) { return safe_(function () { return userPublik_(auth_(token, null, true)); }); }
function auth_logout(token) { return safe_(function () { try { const s = auth_(token, null, true); log_(s, 'Logout', ''); } catch (e) {} CacheService.getScriptCache().remove('sess_' + String(token || '')); return true; }); }

function auth_ganti(token, lama, baru) {
  return safe_(function () {
    const s = auth_(token, null, true);
    cekPasswordBaru_(baru);
    withLock_(function () {
      const u = table_('Users').filter(function (r) { return r.Username === s.username; })[0];
      if (!u) fail_('Akun tidak ditemukan.');
      if (hash_(String(lama || ''), u.Salt) !== u.Password_Hash) fail_('Password lama tidak sesuai.');
      if (String(lama) === String(baru)) fail_('Password baru harus berbeda dari password lama.');
      const salt = Utilities.getUuid();
      u.Salt = salt; u.Password_Hash = hash_(String(baru), salt); u.Wajib_Ganti = 'Tidak';
      setRow_('Users', u._row, u);
    });
    cabutSesi_(s.username);
    s.iat = Date.now() + 1; s.wajib = false;
    CacheService.getScriptCache().put('sess_' + token, JSON.stringify(s), SESI_TTL);
    log_(s, 'Ganti password', '');
    return userPublik_(s);
  }, true);
}

function sanitizeHtml_(html) {
  html = String(html || '');
  if (html.length > 45000) fail_('Isi terlalu panjang (maksimal 45.000 karakter). Pecah menjadi beberapa tulisan.');
  html = html.replace(/<!--[\s\S]*?-->/g, '')
    .replace(/<(script|style|iframe|object|embed|form|link|meta|svg|math|noscript|template)\b[\s\S]*?<\/\1\s*>/gi, '')
    .replace(/<\/?(script|style|iframe|object|embed|form|link|meta|svg|math|noscript|template)\b[^>]*>/gi, '');
  const OK = { p: [], br: [], strong: [], b: [], em: [], i: [], u: [], h2: [], h3: [], h4: [], ul: [], ol: [], li: [], blockquote: [], a: ['href'], img: ['src', 'alt'], figure: [], figcaption: [], table: [], thead: [], tbody: [], tr: [], th: [], td: [], hr: [], span: [], div: [], sub: [], sup: [] };
  return html.replace(/<\/?([a-zA-Z][a-zA-Z0-9]*)\b([^>]*)>/g, function (m, tag, attrs) {
    tag = tag.toLowerCase();
    if (!OK[tag]) return '';
    if (m.charAt(1) === '/') return '</' + tag + '>';
    let out = '<' + tag;
    OK[tag].forEach(function (a) {
      const mm = attrs.match(new RegExp('(?:^|[\\s"\'])' + a + '\\s*=\\s*(?:"([^"]*)"|\'([^\']*)\'|([^\\s>]+))', 'i'));
      if (!mm) return;
      let v = (mm[1] || mm[2] || mm[3] || '').trim();
      if (a === 'href' && !/^(https?:\/\/|mailto:|tel:|#)/i.test(v)) return;
      if (a === 'src' && !/^https:\/\//i.test(v)) return;
      out += ' ' + a + '="' + v.replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;') + '"';
    });
    if (tag === 'a') out += ' target="_blank" rel="noopener noreferrer"';
    return out + (tag === 'br' || tag === 'hr' || tag === 'img' ? ' />' : '>');
  });
}

function getSubfolder_(name) {
  const props = PropertiesService.getScriptProperties(), key = 'FOLDER_' + name;
  const id = props.getProperty(key);
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  let root;
  try { root = DriveApp.getFolderById(DRIVE_FOLDER_ID); }
  catch (e) { fail_('Folder Google Drive tidak dapat diakses. Pastikan akun pemilik script memiliki akses Editor ke folder dokumen.'); }
  const it = root.getFoldersByName(name);
  const f = it.hasNext() ? it.next() : root.createFolder(name);
  props.setProperty(key, f.getId());
  return f;
}

const MIME_GAMBAR = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
const MIME_DOKUMEN = ['application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', 'application/vnd.ms-excel', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', 'application/vnd.ms-powerpoint', 'application/vnd.openxmlformats-officedocument.presentationml.presentation'];

function magicOk_(b, mime) {
  const h = function (i) { return (b[i] & 0xff); };
  if (mime === 'image/jpeg') return h(0) === 0xFF && h(1) === 0xD8;
  if (mime === 'image/png') return h(0) === 0x89 && h(1) === 0x50 && h(2) === 0x4E && h(3) === 0x47;
  if (mime === 'image/gif') return h(0) === 0x47 && h(1) === 0x49 && h(2) === 0x46;
  if (mime === 'image/webp') return h(0) === 0x52 && h(1) === 0x49 && h(2) === 0x46 && h(3) === 0x46;
  if (mime === 'application/pdf') return h(0) === 0x25 && h(1) === 0x50 && h(2) === 0x44 && h(3) === 0x46;
  if (/openxml/.test(mime)) return h(0) === 0x50 && h(1) === 0x4B;
  return h(0) === 0xD0 && h(1) === 0xCF;
}

// opts: {folder, allowed[], max, prefix, publik}
function simpanBerkas_(file, opts) {
  if (!file || !file.bytes) fail_('Berkas tidak terbaca. Silakan pilih ulang berkas.');
  const mime = str_(file.mimeType, 100).toLowerCase();
  if (opts.allowed.indexOf(mime) < 0) fail_('Jenis berkas tidak diizinkan. Gunakan: ' + (opts.allowed === MIME_GAMBAR ? 'JPG, PNG, WEBP, atau GIF.' : opts.labelJenis || 'format yang didukung.'));
  const bytes = Utilities.base64Decode(String(file.bytes));
  if (bytes.length > opts.max) fail_('Ukuran berkas melebihi batas ' + Math.round(opts.max / 1048576) + ' MB.');
  if (bytes.length < 16 || !magicOk_(bytes, mime)) fail_('Isi berkas tidak sesuai dengan jenisnya.');
  const nama = (opts.prefix || '') + str_(file.name, 120).replace(/[^\w.\- ]/g, '_').slice(0, 80);
  const f = getSubfolder_(opts.folder).createFile(Utilities.newBlob(bytes, mime, nama));
  let warn = '';
  if (opts.publik) { try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) { warn = 'Berkas tersimpan, tetapi pengaturan akses publik ditolak oleh kebijakan Google Drive Anda.'; } }
  const id = f.getId();
  return { id: id, mime: mime, name: nama, warn: warn, url: mime.indexOf('image/') === 0 ? 'https://lh3.googleusercontent.com/d/' + id : 'https://drive.google.com/file/d/' + id + '/view' };
}
function bacaBerkasPrivat_(id) {
  if (!id) fail_('Berkas tidak ada.');
  const blob = DriveApp.getFileById(id).getBlob();
  return { mime: blob.getContentType(), name: blob.getName(), dataUrl: 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes()) };
}

function jadwalHari_(c, d) {
  const j = jsonSafe_(c.Jadwal_Kunjungan, {});

function liburMap_(c) {
  const m = {};
  barisList_(c.Libur_Kunjungan).forEach(function (l) {
    const i = l.indexOf('|'), d = (i < 0 ? l : l.slice(0, i)).trim();
    if (isDate_(d)) m[d] = (i < 0 ? '' : l.slice(i + 1).trim()) || 'Kunjungan ditiadakan';
  });

}
function aktifKunjungan_(s) { return s === 'Menunggu Verifikasi' || s === 'Disetujui'; }
function terpakai_(rows) {
  const u = {};
  rows.forEach(function (r) { if (aktifKunjungan_(r.Status)) { const k = r.Tgl_Kunjungan + '|' + r.Sesi; u[k] = (u[k] || 0) + 1; } });
  return u;
}
const HARI = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];

function publik_jadwal() {
  return safe_(function () {
    const c = cfg_(), libur = liburMap_(c), used = terpakai_(table_('Kunjungan'));
    const min = Math.max(0, int_(c.Min_Hari_Daftar, 1)), max = Math.min(60, Math.max(min, int_(c.Maks_Hari_Daftar, 14))), t = today_(), days = [];
    for (let i = min; i <= max; i++) {
      const d = addDays_(t, i), sesi = jadwalHari_(c, d);
      if (libur[d]) { days.push({ tanggal: d, hari: HARI[dow_(d)], libur: libur[d], sesi: [] }); continue; }
      if (!sesi.length) continue;
      days.push({ tanggal: d, hari: HARI[dow_(d)], libur: '', sesi: sesi.map(function (s) { const kuota = int_(s.kuota, 0); return { nama: s.nama, jam: s.jam || '', kuota: kuota, sisa: Math.max(0, kuota - (used[d + '|' + s.nama] || 0)) }; }) });
    }
    const jadwalMingguan = [1, 2, 3, 4, 5, 6, 0].map(function (n) { const a = jsonSafe_(c.Jadwal_Kunjungan, {})[String(n)]; return { hari: HARI[n], sesi: Array.isArray(a) ? a.filter(function (s) { return s && s.nama; }) : [] }; });
    return {
      dibuka: c.Pendaftaran_Dibuka !== 'Tidak', days: days, mingguan: jadwalMingguan,
      aturan: { min: min, max: max, maksPengikut: Math.max(0, int_(c.Maks_Pengikut, 3)), ktpWajib: c.KTP_Wajib === 'Ya' },
      fasilitas: barisList_(c.Fasilitas_Disabilitas)
    };
  });
}

function cekSlot_(c, tgl, namaSesi, rows, abaikanNoReg, petugas) {
  if (!isDate_(tgl)) fail_('Tanggal kunjungan tidak valid.');
  const t = today_(), min = petugas ? 0 : Math.max(0, int_(c.Min_Hari_Daftar, 1)), max = petugas ? 120 : Math.max(min, int_(c.Maks_Hari_Daftar, 14));
  if (tgl < addDays_(t, min)) fail_(min === 0 ? 'Tanggal kunjungan tidak boleh sudah lewat.' : 'Pendaftaran paling lambat ' + min + ' hari sebelum tanggal kunjungan.');
  if (tgl > addDays_(t, max)) fail_('Pendaftaran hanya dapat dilakukan maksimal ' + max + ' hari ke depan.');
  const libur = liburMap_(c);
  if (libur[tgl]) fail_('Tanggal tersebut tidak tersedia: ' + libur[tgl] + '.');
  const sesi = jadwalHari_(c, tgl).filter(function (s) { return s.nama === namaSesi; })[0];
  if (!sesi) fail_('Sesi kunjungan yang dipilih tidak tersedia pada tanggal tersebut.');
  const used = terpakai_(rows.filter(function (r) { return r.No_Reg !== abaikanNoReg; }));
  if ((used[tgl + '|' + sesi.nama] || 0) >= int_(sesi.kuota, 0)) fail_('Kuota sesi tersebut sudah penuh. Silakan pilih sesi atau tanggal lain.');
  return sesi;
}

    const nama = str_(f.nama, 100); if (nama.length < 3) fail_('Nama lengkap wajib diisi (minimal 3 huruf).');
    const nik = String(f.nik || '').replace(/\D/g, ''); if (nik.length !== 16) fail_('NIK harus terdiri dari 16 digit angka.');
    const hp = normHp_(f.hp); if (!/^0\d{8,13}$/.test(hp)) fail_('Nomor HP / WhatsApp tidak valid.');
    const email = str_(f.email, 120).toLowerCase(); if (email && !validEmail_(email)) fail_('Format email tidak valid.');
    const alamat = str_(f.alamat, 300); if (alamat.length < 5) fail_('Alamat wajib diisi.');
    const wbp = str_(f.namaWbp, 120); if (wbp.length < 3) fail_('Nama warga binaan yang dikunjungi wajib diisi.');
    const hub = str_(f.hubungan, 60); if (HUBUNGAN.indexOf(hub) < 0) fail_('Pilih hubungan dengan warga binaan.');
    if (f.setuju !== true) fail_('Anda harus menyetujui pernyataan kebenaran data dan kebijakan privasi.');
    const maksP = Math.max(0, int_(c.Maks_Pengikut, 3));
    const pengikut = (Array.isArray(f.pengikut) ? f.pengikut : []).slice(0, 20).map(function (p) { return { nama: str_(p && p.nama, 100), hubungan: str_(p && p.hubungan, 60), usia: str_(p && p.usia, 3) }; }).filter(function (p) { return p.nama; });
    if (pengikut.length > maksP) fail_('Jumlah pengikut maksimal ' + maksP + ' orang.');
    pengikut.forEach(function (p) { if (p.nama.length < 3) fail_('Nama pengikut minimal 3 huruf.'); });
    const kebutuhan = (Array.isArray(f.kebutuhan) ? f.kebutuhan : []).filter(function (k) { return KEBUTUHAN_KHUSUS.indexOf(k) >= 0; });
    const detail = str_(f.detailKebutuhan, 400);
    if (c.KTP_Wajib === 'Ya' && !(berkas && berkas.bytes)) fail_('Foto identitas (KTP) wajib diunggah.');
    rateLimit_('rl_kj_' + nik, 5, 86400, 'Pendaftaran dengan NIK ini sudah mencapai batas harian. Silakan coba besok atau hubungi petugas.');
    cekSlot_(c, str_(f.tgl, 10), str_(f.sesi, 60), table_('Kunjungan'));   // pra-cek cepat sebelum unggah berkas

    let berkasId = '';
    if (berkas && berkas.bytes) berkasId = simpanBerkas_(berkas, { folder: 'Berkas_Kunjungan_PRIVAT', allowed: MIME_GAMBAR.concat(['application/pdf']), max: 4 * 1024 * 1024, prefix: 'KTP_', publik: false, labelJenis: 'JPG, PNG, WEBP, atau PDF.' }).id;

    let hasil;
    try {
      hasil = withLock_(function () {
        const rows = table_('Kunjungan');
        const sesi = cekSlot_(c, str_(f.tgl, 10), str_(f.sesi, 60), rows);
        rows.forEach(function (r) {
          if (aktifKunjungan_(r.Status) && r.Tgl_Kunjungan === f.tgl && r.NIK === nik) fail_('NIK ini sudah terdaftar pada tanggal tersebut (No. Reg ' + r.No_Reg + ').');
        });
        const maksW = int_(c.Maks_Kunjungan_WBP_Hari, 0);
        if (maksW > 0) {
          const n = rows.filter(function (r) { return aktifKunjungan_(r.Status) && r.Tgl_Kunjungan === f.tgl && r.Nama_WBP.toLowerCase() === wbp.toLowerCase(); }).length;
          if (n >= maksW) fail_('Warga binaan tersebut sudah mencapai batas kunjungan pada tanggal itu. Silakan pilih tanggal lain.');
        }
        const noReg = kodeUnik_('KJ', rows, 'No_Reg'), kode = randCode_(6);
        addRow_('Kunjungan', { No_Reg: noReg, Kode_Cek: kode, Tgl_Daftar: now_(), Tgl_Kunjungan: f.tgl, Sesi: sesi.nama, Jam: sesi.jam || '', Nama: nama, NIK: nik, No_HP: hp, Email: email, Alamat: alamat, Nama_WBP: wbp, Hubungan: hub, Pengikut: JSON.stringify(pengikut), Kebutuhan: JSON.stringify(kebutuhan), Detail_Kebutuhan: detail, Berkas_ID: berkasId, Status: 'Menunggu Verifikasi', Catatan: '', Diproses_Oleh: '', Waktu_Proses: '' });
        return { noReg: noReg, kode: kode, tgl: f.tgl, sesi: sesi.nama, jam: sesi.jam || '' };
      });
    } catch (e) {
      if (berkasId) { try { DriveApp.getFileById(berkasId).setTrashed(true); } catch (x) {} }
      throw e;
    }
    if (email) kirimEmail_(email, 'Pendaftaran kunjungan diterima: ' + hasil.noReg, '<p>Yth. <b>' + esc_(nama) + '</b>,</p><p>Pendaftaran kunjungan Anda telah kami terima dan sedang menunggu verifikasi petugas.</p><table cellpadding="6" style="border-collapse:collapse"><tr><td>No. Registrasi</td><td><b>' + esc_(hasil.noReg) + '</b></td></tr><tr><td>Kode Cek</td><td><b>' + esc_(hasil.kode) + '</b></td></tr><tr><td>Tanggal</td><td>' + esc_(hasil.tgl) + '</td></tr><tr><td>Sesi</td><td>' + esc_(hasil.sesi + ' ' + hasil.jam) + '</td></tr></table><p>Simpan nomor registrasi dan kode cek untuk memeriksa status pendaftaran di website.</p>');
    if (c.Email_Notifikasi_Layanan) kirimEmail_(c.Email_Notifikasi_Layanan, 'Pendaftaran kunjungan baru ' + hasil.noReg, '<p>Ada pendaftaran kunjungan baru untuk tanggal <b>' + esc_(hasil.tgl) + '</b> (' + esc_(hasil.sesi) + '). Silakan verifikasi melalui dashboard petugas.</p>');
    return hasil;
  });
}

function cariKunjungan_(noReg, kode) {
  noReg = str_(noReg, 40).toUpperCase(); kode = str_(kode, 10).toUpperCase();
  if (!noReg || !kode) fail_('Nomor registrasi dan kode cek wajib diisi.');
  const c = CacheService.getScriptCache(), fk = 'rl_cekf_' + noReg;
  if (int_(c.get(fk), 0) >= 8) fail_('Terlalu banyak percobaan. Silakan coba lagi dalam satu jam.');
  const r = table_('Kunjungan').filter(function (x) { return x.No_Reg === noReg; })[0];
  if (!r || r.Kode_Cek !== kode) { c.put(fk, String(int_(c.get(fk), 0) + 1), 3600); fail_('Data tidak ditemukan. Periksa kembali nomor registrasi dan kode cek Anda.'); }
  return r;
}
function publik_cekKunjungan(noReg, kode) {
  return safe_(function () {
    const r = cariKunjungan_(noReg, kode);
    return {
      noReg: r.No_Reg, status: r.Status, tglDaftar: r.Tgl_Daftar, tgl: r.Tgl_Kunjungan, sesi: r.Sesi, jam: r.Jam, nama: r.Nama, namaWbp: r.Nama_WBP,
      pengikut: jsonSafe_(r.Pengikut, []).length, kebutuhan: jsonSafe_(r.Kebutuhan, []), catatan: r.Catatan,
      dapatBatal: aktifKunjungan_(r.Status) && r.Tgl_Kunjungan >= today_()
    };
  });
}
function publik_batalKunjungan(noReg, kode) {
  return safe_(function () {
    cariKunjungan_(noReg, kode);
    return withLock_(function () {
      const r = table_('Kunjungan').filter(function (x) { return x.No_Reg === String(noReg).trim().toUpperCase(); })[0];
      if (!r || !aktifKunjungan_(r.Status) || r.Tgl_Kunjungan < today_()) fail_('Pendaftaran ini tidak dapat dibatalkan.');
      r.Status = 'Dibatalkan'; r.Catatan = 'Dibatalkan oleh pengunjung pada ' + now_(); r.Waktu_Proses = now_();
      setRow_('Kunjungan', r._row, r);
      return true;
    });
  });
}

    f = f || {};
    if (str_(f.website, 50)) fail_('Permintaan tidak valid.');
    const jenis = f.jenis === 'Permohonan Informasi' ? 'Permohonan Informasi' : 'Pengaduan';
    const anonim = f.anonim === true && jenis === 'Pengaduan';
    const nama = anonim ? '' : str_(f.nama, 100);
    const kontak = str_(f.kontak, 120);
    if (!anonim && nama.length < 3) fail_('Nama wajib diisi, atau pilih kirim secara anonim.');
    if (jenis === 'Permohonan Informasi' && !kontak) fail_('Kontak (email atau nomor HP) wajib diisi agar kami dapat menyampaikan jawaban.');
    if (kontak && !validEmail_(kontak.toLowerCase()) && !/^\+?[\d\s\-]{8,16}$/.test(kontak)) fail_('Kontak harus berupa email atau nomor HP yang valid.');
    const kategori = jenis === 'Pengaduan' ? str_(f.kategori, 80) : 'Informasi Publik';

    rateLimit_('rl_pgd_all', 80, 3600, 'Layanan sedang menerima banyak kiriman. Silakan coba lagi nanti.');

function admin_dashboard(token) {
  return safe_(function () {
    const s = auth_(token), t = today_(), out = { role: s.role };
    if (s.role === R_SUPER || s.role === R_LAYANAN) {
      const k = table_('Kunjungan'), besok = addDays_(t, 1);
      const seri = [];
      for (let i = 13; i >= 0; i--) { const d = addDays_(t, -i); seri.push({ tanggal: d, jumlah: k.filter(function (r) { return r.Tgl_Daftar.slice(0, 10) === d; }).length }); }
      out.kunjungan = {
        menunggu: k.filter(function (r) { return r.Status === 'Menunggu Verifikasi'; }).length,
        hariIni: k.filter(function (r) { return r.Tgl_Kunjungan === t && r.Status === 'Disetujui'; }).length,
        besok: k.filter(function (r) { return r.Tgl_Kunjungan === besok && r.Status === 'Disetujui'; }).length,
        bulanIni: k.filter(function (r) { return r.Tgl_Daftar.slice(0, 7) === t.slice(0, 7); }).length,
        kebutuhanKhusus: k.filter(function (r) { return aktifKunjungan_(r.Status) && r.Tgl_Kunjungan >= t && jsonSafe_(r.Kebutuhan, []).length > 0; }).length,
        seri: seri
      };
    }
    if (s.role === R_SUPER || s.role === R_HUMAS) {
      const b = table_('Berita'), p = table_('Pengaduan'), cnt = {};
      STATUS_BERITA.forEach(function (st) { cnt[st] = b.filter(function (x) { return x.Status === st; }).length; });
      out.berita = cnt;
      out.pengaduan = { baru: p.filter(function (x) { return x.Status === 'Baru' && x.Jenis === 'Pengaduan'; }).length, permohonanBaru: p.filter(function (x) { return x.Status === 'Baru' && x.Jenis === 'Permohonan Informasi'; }).length, diproses: p.filter(function (x) { return x.Status === 'Diproses'; }).length };
    }
    if (s.role === R_SUPER) {
      out.pengguna = table_('Users').filter(function (u) { return u.Status === 'Aktif'; }).length;
      out.log = table_('Log').slice(-8).reverse();
    }
    return out;
  }, true);
}

function ringkasKunjungan_(r) {
  return { noReg: r.No_Reg, tglDaftar: r.Tgl_Daftar, tgl: r.Tgl_Kunjungan, sesi: r.Sesi, jam: r.Jam, nama: r.Nama, namaWbp: r.Nama_WBP, status: r.Status, pengikut: jsonSafe_(r.Pengikut, []).length, khusus: jsonSafe_(r.Kebutuhan, []).length > 0 };
}
function admin_kunjunganList(token, flt) {
  return safe_(function () {
    auth_(token, [R_LAYANAN]);
    flt = flt || {};
    const q = str_(flt.q, 60).toLowerCase(), st = str_(flt.status, 40), dari = str_(flt.dari, 10), sampai = str_(flt.sampai, 10);
    let rows = table_('Kunjungan');
    if (st) rows = rows.filter(function (r) { return r.Status === st; });
    if (isDate_(dari)) rows = rows.filter(function (r) { return r.Tgl_Kunjungan >= dari; });
    if (isDate_(sampai)) rows = rows.filter(function (r) { return r.Tgl_Kunjungan <= sampai; });
    if (flt.khusus) rows = rows.filter(function (r) { return jsonSafe_(r.Kebutuhan, []).length > 0; });
    if (q) rows = rows.filter(function (r) { return (r.No_Reg + ' ' + r.Nama + ' ' + r.Nama_WBP).toLowerCase().indexOf(q) >= 0; });
    rows.sort(function (a, b) { return b.Tgl_Daftar.localeCompare(a.Tgl_Daftar); });
    return { total: rows.length, items: rows.slice(0, 300).map(ringkasKunjungan_) };
  }, true);
}

function admin_kunjunganProses(token, noReg, status, catatan) {
  return safe_(function () {
    const s = auth_(token, [R_LAYANAN]);
    status = str_(status, 40); catatan = str_(catatan, 400);
    if (['Disetujui', 'Ditolak', 'Selesai', 'Tidak Hadir'].indexOf(status) < 0) fail_('Status tidak valid.');
    if (status === 'Ditolak' && catatan.length < 5) fail_('Alasan penolakan wajib diisi (minimal 5 karakter).');
    const hasil = withLock_(function () {
      const r = table_('Kunjungan').filter(function (x) { return x.No_Reg === str_(noReg, 40); })[0];
      if (!r) fail_('Data pendaftaran tidak ditemukan.');
      if (status === 'Disetujui' && r.Status !== 'Menunggu Verifikasi') fail_('Hanya pendaftaran berstatus Menunggu Verifikasi yang dapat disetujui.');
      if (status === 'Disetujui' && r.Tgl_Kunjungan < today_()) fail_('Tanggal kunjungan sudah lewat. Tolak pendaftaran atau jadwalkan ulang.');
      if ((status === 'Selesai' || status === 'Tidak Hadir') && r.Status !== 'Disetujui') fail_('Kehadiran hanya dapat dicatat untuk pendaftaran yang sudah disetujui.');
      if (status === 'Ditolak' && ['Menunggu Verifikasi', 'Disetujui'].indexOf(r.Status) < 0) fail_('Pendaftaran ini sudah final dan tidak dapat ditolak.');
      r.Status = status; r.Catatan = catatan || r.Catatan; r.Diproses_Oleh = s.nama; r.Waktu_Proses = now_();
      setRow_('Kunjungan', r._row, r);
      return clean_(r);
    });
    log_(s, 'Kunjungan ' + status, hasil.No_Reg);
    if ((status === 'Disetujui' || status === 'Ditolak') && hasil.Email) {
      kirimEmail_(hasil.Email, 'Pendaftaran kunjungan ' + hasil.No_Reg + ' ' + status.toLowerCase(), '<p>Yth. <b>' + esc_(hasil.Nama) + '</b>,</p><p>Pendaftaran kunjungan Anda untuk tanggal <b>' + esc_(hasil.Tgl_Kunjungan) + '</b> (' + esc_(hasil.Sesi + ' ' + hasil.Jam) + ') telah <b>' + esc_(status.toLowerCase()) + '</b>.</p>' + (hasil.Catatan ? '<p>Catatan petugas: ' + esc_(hasil.Catatan) + '</p>' : '') + (status === 'Disetujui' ? '<p>Mohon hadir tepat waktu dan membawa identitas asli sesuai pendaftaran.</p>' : ''));
    }
    return true;
  }, true);
}

function bolehStatusBerita_(role, perlu, dari, ke) {
  if (role === R_SUPER) return true;
  if (dari === ke) return true;
  if (ke === 'Menunggu Persetujuan') return dari === 'Draft';
  if (ke === 'Arsip') return dari === 'Draft' || dari === 'Menunggu Persetujuan' || (!perlu && (dari === 'Terbit' || dari === 'Terjadwal'));
  if (ke === 'Draft') return dari === 'Arsip' || dari === 'Menunggu Persetujuan' || (!perlu && (dari === 'Terbit' || dari === 'Terjadwal'));
  if (ke === 'Terbit' || ke === 'Terjadwal') return !perlu;

function admin_save(token, ent, d) {

function normJadwal_(raw) {
  const j = typeof raw === 'string' ? jsonSafe_(raw, null) : raw;
  if (!j || typeof j !== 'object') fail_('Format jadwal kunjungan tidak valid.');
  const out = {};
  for (let h = 0; h <= 6; h++) {
    const arr = Array.isArray(j[String(h)]) ? j[String(h)] : [], seen = {}, list = [];
    if (arr.length > 8) fail_('Maksimal 8 sesi per hari.');
    arr.forEach(function (x) {
      const nama = str_(x && x.nama, 40), jam = str_(x && x.jam, 30), kuota = int_(x && x.kuota, -1);
      if (!nama) return;
      if (kuota < 0 || kuota > 500) fail_('Kuota sesi "' + nama + '" harus berupa angka 0 sampai 500.');
      if (seen[nama.toLowerCase()]) fail_('Nama sesi "' + nama + '" ganda pada hari ' + HARI[h] + '.');
      seen[nama.toLowerCase()] = 1; list.push({ nama: nama, jam: jam, kuota: kuota });
    });
    if (list.length) out[String(h)] = list;
  }
  return JSON.stringify(out);
}
function validSetting_(k, v) {
  v = v === undefined || v === null ? '' : v;
  if (/^(Logo_URL|Hero_URL|Kepala_Foto|Maps_Link|Sosmed_)/.test(k)) return validUrl_(v, true);
  if (k === 'Maps_Embed') { v = str_(v, 1200); if (v && !/^https:\/\/www\.google\.com\/maps\/embed/.test(v)) fail_('URL embed peta harus diawali https://www.google.com/maps/embed'); return v; }
  if (k === 'Pendaftaran_Dibuka' || k === 'KTP_Wajib' || k === 'Wajib_Persetujuan_Berita') return (v === 'Ya' || v === true) ? 'Ya' : 'Tidak';
  if (k === 'Maks_Hari_Daftar') return String(Math.min(60, Math.max(1, int_(v, 14))));
  if (k === 'Min_Hari_Daftar') return String(Math.min(30, Math.max(0, int_(v, 1))));
  if (k === 'Maks_Pengikut') return String(Math.min(10, Math.max(0, int_(v, 3))));
  if (k === 'Maks_Kunjungan_WBP_Hari') return String(Math.min(50, Math.max(0, int_(v, 0))));
  if (k === 'WhatsApp_Layanan') { v = String(v).replace(/\D/g, ''); if (v && !/^62\d{8,13}$/.test(v)) fail_('Nomor WhatsApp harus berformat 62xxxxxxxxxx.'); return v; }
  if (k === 'Jadwal_Kunjungan') return normJadwal_(v);
  if (k === 'Libur_Kunjungan') { return barisList_(v).map(function (l) { const i = l.indexOf('|'), d = (i < 0 ? l : l.slice(0, i)).trim(); if (!isDate_(d)) fail_('Format tanggal libur salah: "' + l.slice(0, 30) + '". Gunakan yyyy-mm-dd|keterangan'); return d + '|' + (i < 0 ? '' : str_(l.slice(i + 1), 80)); }).join('\n'); }
  if (/^Email/.test(k) && k !== 'Email') { v = str_(v, 120); if (v && !validEmail_(v)) fail_('Format email tidak valid.'); return v; }
  if (k === 'Email') { v = str_(v, 120); if (v && !validEmail_(v)) fail_('Format email resmi tidak valid.'); return v; }
  const long = ['Kepala_Sambutan', 'Fasilitas_Disabilitas'].indexOf(k) >= 0;
  return str_(v, long ? 4000 : 300);
}
function admin_simpanPengaturan(token, obj) {
  return safe_(function () {
    const s = auth_(token), boleh = kunciBolehDiubah_(s.role);
    obj = obj || {};
    const rec = {};
    Object.keys(obj).forEach(function (k) {
      if (boleh.indexOf(k) < 0) fail_('Anda tidak berhak mengubah pengaturan "' + k + '".', 'AKSES');
      rec[k] = validSetting_(k, obj[k]);
    });
    withLock_(function () {
      const rows = table_('Pengaturan');
      Object.keys(rec).forEach(function (k) {
        const ex = rows.filter(function (r) { return r.Kunci === k; })[0], ket = (DEFAULT_SETTINGS.filter(function (d) { return d[0] === k; })[0] || [])[2] || '';
        if (ex) { ex.Nilai = rec[k]; setRow_('Pengaturan', ex._row, clean_(ex)); } else addRow_('Pengaturan', { Kunci: k, Nilai: rec[k], Keterangan: ket });
      });
    });
    bersihCache_();
    log_(s, 'Ubah pengaturan', Object.keys(rec).join(', '));
    return true;
  }, true);
}

function admin_userSimpan(token, d) {
  return safe_(function () {
    const s = auth_(token, [R_SUPER]);
    d = d || {};
    const username = str_(d.username, 30).toLowerCase(), nama = str_(d.nama, 100), jabatan = str_(d.jabatan, 100), role = str_(d.role, 30), email = str_(d.email, 120).toLowerCase();
    if (!/^[a-z0-9._-]{3,30}$/.test(username)) fail_('Username 3-30 karakter: huruf kecil, angka, titik, strip, atau garis bawah.');
    if (nama.length < 3) fail_('Nama wajib diisi.');
    if ([R_SUPER, R_LAYANAN, R_HUMAS].indexOf(role) < 0) fail_('Pilih peran akun.');
    if (email && !validEmail_(email)) fail_('Format email tidak valid.');
    const status = d.status === 'Nonaktif' ? 'Nonaktif' : 'Aktif';
    withLock_(function () {
      const rows = table_('Users'), ex = rows.filter(function (r) { return r.Username === username; })[0];
      const aktifSuper = rows.filter(function (r) { return r.Role === R_SUPER && r.Status === 'Aktif' && r.Username !== username; }).length;
      if (ex) {
        if (ex.Role === R_SUPER && (role !== R_SUPER || status !== 'Aktif') && aktifSuper < 1) fail_('Harus tersisa minimal satu Super Admin aktif.');
        if (ex.Username === s.username && status !== 'Aktif') fail_('Anda tidak dapat menonaktifkan akun Anda sendiri.');
        ex.Nama = nama; ex.Jabatan = jabatan; ex.Role = role; ex.Status = status; ex.Email = email;
        setRow_('Users', ex._row, clean_(ex));
        if (status !== 'Aktif' || ex.Role !== role) cabutSesi_(username);
      } else {
        cekPasswordBaru_(d.password);
        const salt = Utilities.getUuid();
        addRow_('Users', { Username: username, Nama: nama, Jabatan: jabatan, Role: role, Password_Hash: hash_(String(d.password), salt), Salt: salt, Status: status, Wajib_Ganti: 'Ya', Email: email, Terakhir_Login: '', Dibuat: now_() });
      }
    });
    log_(s, 'Simpan pengguna', username);
    return true;
  }, true);
}
function admin_userResetPassword(token, username, passwordBaru) {
  return safe_(function () {
    const s = auth_(token, [R_SUPER]);
    cekPasswordBaru_(passwordBaru);
    withLock_(function () {
      const u = table_('Users').filter(function (r) { return r.Username === str_(username, 30); })[0];
      if (!u) fail_('Akun tidak ditemukan.');
      const salt = Utilities.getUuid();
      u.Salt = salt; u.Password_Hash = hash_(String(passwordBaru), salt); u.Wajib_Ganti = 'Ya';
      setRow_('Users', u._row, clean_(u));
    });
    cabutSesi_(str_(username, 30));
    CacheService.getScriptCache().remove('fail_' + str_(username, 30).toLowerCase());
    log_(s, 'Reset password', username);
    return true;
  }, true);
}

// 1) Ubah dua nilai di bawah, 2) pilih fungsi resetPasswordDarurat, 3) klik Run, 4) HAPUS password dari kode setelah selesai.
function resetPasswordDarurat() {
  const USERNAME = 'admin';
  const PASSWORD_BARU = 'GantiDenganPasswordBaru1';   // minimal 8 karakter, huruf dan angka
  cekPasswordBaru_(PASSWORD_BARU);
  withLock_(function () {
    const u = table_('Users').filter(function (r) { return r.Username === USERNAME; })[0];
    if (!u) throw new Error('Akun tidak ditemukan: ' + USERNAME);
    const salt = Utilities.getUuid();
    u.Salt = salt; u.Password_Hash = hash_(PASSWORD_BARU, salt); u.Wajib_Ganti = 'Ya'; u.Status = 'Aktif';
    setRow_('Users', u._row, clean_(u));
  });
  cabutSesi_(USERNAME);
  CacheService.getScriptCache().remove('fail_' + USERNAME.toLowerCase());
  Logger.log('Password akun ' + USERNAME + ' sudah direset. Wajib diganti saat login berikutnya.');
}
